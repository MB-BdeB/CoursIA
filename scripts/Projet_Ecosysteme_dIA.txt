def watchdog_logger(temps, temp_gpu, errors_ecc):   
    print(f"[LOG - Sec {temps}] Température GPU: {temp_gpu}°C 
| Erreurs ECC: {errors_ecc}")   
       
    # RÈGLE 1 : Si la température dépasse 82°C, lever une 
alerte de "Thermal Throttling"

        print("  --> [ALERT SYSTEM] Vitesse d'horloge bridée 
automatiquement (Thermal Throttling) !")
